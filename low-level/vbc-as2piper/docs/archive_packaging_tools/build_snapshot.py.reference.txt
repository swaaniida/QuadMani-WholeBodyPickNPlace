"""Create a non-destructive, allowlisted source/artifact snapshot for GitHub."""
from pathlib import Path
import os,json,hashlib,shutil,gzip,re,datetime
ROOT=Path(__file__).resolve().parents[1];OUT=ROOT/'github-export/low-level/vbc-as2piper'
assert not OUT.exists(),'Refuse to overwrite an existing export'
OUT.mkdir(parents=True);records=[];excluded=[]
DENY={'.git','.venv','venv','node_modules','__pycache__','.pytest_cache','.torch_extensions','.cache','.deps','.wandb_deps','wandb','launch_logs','backups'}
FORBIDDEN={'.netrc','.env','auth.json','credentials.json','id_rsa','id_ed25519'}
secret=re.compile(rb'(?:gh[pousr]_[A-Za-z0-9]{30,}|github_pat_[A-Za-z0-9_]{30,}|-----BEGIN (?:RSA |OPENSSH |EC )?PRIVATE KEY-----)')
def add(src,dst,category='source'):
 src=Path(src);dst=OUT/dst
 if src.is_symlink():src=src.resolve()
 if not src.is_file():return
 if src.name in FORBIDDEN or src.suffix.lower() in {'.pem','.key','.wandb','.pyc','.so','.o','.log'}:return
 data=src.read_bytes()
 if secret.search(data):excluded.append({'path':str(src),'reason':'secret-pattern; content never printed'});return
 dst.parent.mkdir(parents=True,exist_ok=True);dst.write_bytes(data)
 records.append({'source':str(src),'path':str(dst.relative_to(OUT)),'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest(),'category':category})
def tree(src,dst,category='source',skip=()):
 src=Path(src)
 if not src.exists():return
 for p in sorted(src.rglob('*')):
  rel=p.relative_to(src)
  if any(x in DENY or x in skip for x in rel.parts):continue
  if p.is_file() and p.suffix.lower() not in {'.pt','.pth','.zip'}:add(p,Path(dst)/rel,category)
variants=['low-level','low-level-as2piper','low-level-as2piper-forward','low-level-as2piper-gait','low-level-as2piper-hw','low-level-as2piper-head-orin','low-level-as2piper-head-orin-nocontact','low-level-as2piper-head-orin-deploy']
for variant in variants:
 for sub in ['legged_gym','rsl_rl_local','resources','force_ablation','validation']:
  tree(ROOT/variant/sub,Path(variant)/sub,skip=('experiments',))
 for p in sorted((ROOT/variant).glob('*')):
  if p.is_file() and p.suffix in ['.py','.sh','.md','.json']:
   if p.name.startswith('restart') or p.name=='baseline_hashes.json':continue
   add(p,Path(variant)/p.name)
 # destructive historical restart helper is not a supported executable
 p=OUT/variant/'force_ablation/restart_push25.py'
 if p.exists():p.rename(p.with_suffix('.py.reference.txt'))
for sub in ['tools','docs','mujoco_mount_compare','wrist_camera_integration/source_bundle','reports']:
 tree(ROOT/sub,sub,'evidence' if sub=='reports' else 'source',skip=('downloads',))
for f in ['LICENCE','README.md','TEAM_HANDOFF_KO.md','AGENTS.md']:
 add(ROOT/f,Path('docs/historical')/f,'historical-document')
tree(ROOT/'third_party/rsl_rl','third_party/rsl_rl','third-party-source')
# Preserve original B1 source and exact reference checkpoint; do not alter it.
legacy=ROOT.parent/'vbc-legacy'
for sub in ['legged_gym','resources']:
 tree(legacy/'low-level'/sub,Path('reference/vbc-legacy/low-level')/sub,'reference-B1')
for f in ['README.md','setup.py']:
 add(legacy/'low-level'/f,Path('reference/vbc-legacy/low-level')/f,'reference-B1')
add(legacy/'low-level/logs/b1z1-low/original-run01/model_45000.pt','reference/vbc-legacy/low-level/logs/b1z1-low/original-run01/model_45000.pt','checkpoint-B1')
tree(ROOT/'third_party/rsl_rl','reference/vbc-legacy/third_party/rsl_rl','reference-B1')
# Source CAD/URDF inputs used by construction scripts.
for sub in ['unitree_ros/robots/as2_description','agx_arm_urdf']:
 tree(ROOT.parent/'robot-sources'/sub,Path('robot-sources')/sub,'robot-source')
for p in (ROOT.parent/'robot-sources/unitree_ros').glob('*LICENSE*'):add(p,Path('robot-sources/unitree_ros')/p.name,'license')
source_orin=ROOT.parent/'deep-wbc-as2piper/as2piper/legged_gym/resources/robots/as2piper/urdf/as2piper_center_umi_orin64.urdf'
add(source_orin,'reference/orin_source/as2piper_center_umi_orin64.urdf','reference-only')
checkpoint_index=[]
# All completed checkpoints for the two principal runs, frozen at discovery time.
for variant in ['low-level-as2piper-head-orin-nocontact','low-level-as2piper-head-orin-deploy']:
 base=ROOT/variant/'experiments/contact_force_ablation'
 for seed in base.glob('seed_*'):
  for p in seed.glob('*'):
   if p.is_file() and p.suffix in ['.json','.pt','.txt']:add(p,Path(variant)/'experiments/contact_force_ablation'/seed.name/p.name,'initialization')
 for run in base.glob('runs/*'):
  files=sorted(run.glob('model_*.pt'),key=lambda p:int(p.stem.split('_')[-1]))
  if not files:continue
  for p in files:
   dst=Path(variant)/'experiments/contact_force_ablation/runs'/run.name/p.name;add(p,dst,'checkpoint')
   checkpoint_index.append({'variant':variant,'run':run.name,'iteration':int(p.stem.split('_')[-1]),'path':str(dst),'source_bytes':p.stat().st_size})
  for name in ['manifest.json','initial_model_sha256.txt']:add(run/name,Path(variant)/'experiments/contact_force_ablation/runs'/run.name/name,'run-provenance')
  metric=run/'ablation_metrics.jsonl'
  if metric.exists():
   dst=OUT/variant/'experiments/contact_force_ablation/runs'/run.name/'ablation_metrics.jsonl.gz';dst.parent.mkdir(parents=True,exist_ok=True)
   # Bound the active file by its starting size and discard a partial final line.
   remaining=metric.stat().st_size
   with metric.open('rb') as inp,gzip.open(dst,'wb',compresslevel=6) as out:
    while remaining:
     line=inp.readline();remaining-=len(line)
     if not line or not line.endswith(b'\n') or remaining<0:break
     out.write(line)
  tree(base/'evaluations',Path(variant)/'experiments/contact_force_ablation/evaluations','evaluation')
  tree(base/'validation',Path(variant)/'experiments/contact_force_ablation/validation','validation')
# Prototype policies: final/latest + important round-number checkpoints per run.
for variant in variants:
 if variant in ['low-level-as2piper-head-orin-deploy','low-level-as2piper-head-orin-nocontact']:continue
 bydir={}
 for p in (ROOT/variant).rglob('model_*.pt'):
  if p.is_file() and re.fullmatch(r'model_\d+\.pt',p.name):bydir.setdefault(p.parent,[]).append(p)
 for folder,files in bydir.items():
  files.sort(key=lambda p:int(p.stem.split('_')[-1]));chosen=[p for p in files if p==files[-1] or int(p.stem.split('_')[-1]) in [0,7000,10000,15000,20000,30000,45000]]
  for p in chosen:
   dst=p.relative_to(ROOT);add(p,dst,'historical-checkpoint');checkpoint_index.append({'variant':variant,'iteration':int(p.stem.split('_')[-1]),'path':str(dst),'role':'prototype/reference'})
  for f in ['manifest.json','args.json','config.json']:
   add(folder/f,(folder/f).relative_to(ROOT),'historical-config')
# The old live directory acquired a camera AFTER training: restore only the
# exported baseline asset from the exact-byte trained asset kept in reports.
trained=ROOT/'reports/deployment_workspace_review_20261114/trained_asset/urdf/as2piper_head_orin_training.urdf'
add(trained,'low-level-as2piper-head-orin-nocontact/resources/robots/as2piper/urdf/as2piper_head_orin_training.urdf','exact-trained-baseline-asset')
# Export-only launcher portability; native Python training sources stay exact.
for variant in variants:
 for p in (OUT/variant).glob('*.sh'):
  s=p.read_text();s=s.replace('export TORCH_EXTENSIONS_DIR="$PWD/../low-level-as2piper-head-orin/.torch_extensions"','export TORCH_EXTENSIONS_DIR="${TORCH_EXTENSIONS_DIR:-$PWD/.cache/torch_extensions}"')
  s=s.replace('export TORCH_EXTENSIONS_DIR="$PWD/.torch_extensions"','export TORCH_EXTENSIONS_DIR="${TORCH_EXTENSIONS_DIR:-$PWD/.cache/torch_extensions}"')
  if 'exec python' in s and 'PYTHONUNBUFFERED' not in s:s=s.replace('umask 0002','umask 0002\nexport PYTHONUNBUFFERED=1',1)
  p.write_text(s)
(OUT/'docs').mkdir(exist_ok=True)
(OUT/'docs/checkpoint_index.json').write_text(json.dumps(checkpoint_index,indent=2))
(OUT/'docs/source_copy_map.json').write_text(json.dumps(records,indent=2))
(OUT/'docs/excluded_files.json').write_text(json.dumps(excluded,indent=2))
(OUT/'PROJECT_SNAPSHOT.json').write_text(json.dumps({'created_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'source':'vbc-as2piper live workspace; source files untouched','principal_checkpoint_count':sum(x['category']=='checkpoint' for x in records),'redistributed_isaacgym':False,'historical_asset_restoration':'baseline200 URDF restored from exact-byte trained_asset in export only'},indent=2))
for p in [OUT,*OUT.rglob('*')]:
 if p.is_symlink():raise RuntimeError('Unexpected symlink '+str(p))
 p.chmod(0o2775 if p.is_dir() else (0o775 if p.suffix=='.sh' else 0o664))
print('Snapshot created:',OUT,'files copied',len(records),'checkpoints',len(checkpoint_index),'secret-pattern exclusions',len(excluded))
